# atomworks 3.0 (released 2026-10-05) rejects an atom array that carries no `charge` annotation, which
# breaks the structure pipeline the design engine featurises through -- so inference/design_ph.py fails
# on a fresh install. foundry/pyproject.toml asks only for `atomworks[ml]>=2.1.1` (no upper bound), so
# a new resolution takes 3.x; this code was written and tested against the 2.x series.
# Reported by @SimonCrouzet in PR #6.
atomworks[ml]>=2.1.1,<3

# pandas 3.0 defaults to Arrow-backed strings, which FLAML 2.6 can't index (ArrowStringArray has no .iloc);
# pin to the 2.x series the model was developed with.
pandas>=2.1,<3

# --- FLAML protonation labeller (labeller/) ---
flaml>=2.6
xgboost>=2.1
lightgbm>=4.6
scikit-learn>=1.8,<1.9   # match the FLAML model pickles (avoids unpickle version warnings)
shap>=0.52            # only for the labeller SHAP/analysis scripts (not required to train)

# --- notebook / plots (inference/design_ph.ipynb) ---
matplotlib>=3.8
seaborn>=0.13        # imported at module load by mpnn.scripts.eval_pkad (the PKAD benchmark)
nbconvert>=7
nbformat>=5
jupyterlab>=4
